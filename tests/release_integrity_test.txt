function fileExists(path: string) {
  try {
    return Deno.statSync(path).isFile;
  } catch {
    return false;
  }
}

Deno.test("browser session credentials survive reloads and never attach to unrelated references", () => {
  const source = Deno.readTextFileSync(`${javascriptRoot}/api.${javascriptExtension}`);
  const helpersSource = source.slice(source.indexOf("const gameSessionCredentials = new Map();"),
    source.indexOf("async function callGameApi("));
  const storage = new Map<string, string>();
  const localStorage = { getItem: (key: string) => storage.get(key) ?? null,
    setItem: (key: string, value: string) => { storage.set(key, value); } };
  const build = (store = localStorage) => new Function("localStorage", "PHYLOSAUR_STORAGE_KEYS",
    `${helpersSource}; return { rememberGameSessionToken, getGameSessionToken, withGameSessionCredentials };`)(
    store, { sessionCredentialPrefix: "credential:" });
  let runtime = build();
  const token = `v1.${"A".repeat(43)}`;
  runtime.rememberGameSessionToken({ sessionId: "session-a", sessionToken: token });
  runtime = build();
  assert(runtime.getGameSessionToken("session-a") === token,
    "Reloading the page must retain the private credential separately from existing UUID saves.");
  const original = { sessionId: "session-a", code: "RAPTOR" };
  const request = runtime.withGameSessionCredentials(original);
  assert(request.sessionToken === token && !Object.hasOwn(original, "sessionToken")
    && !runtime.withGameSessionCredentials({ sessionId: "session-b" }).sessionToken,
    "Only the matching session credential may be attached, without mutating caller payloads.");
  const proof = runtime.withGameSessionCredentials({ proofSessionIds: ["session-a", "session-b"] });
  assert(JSON.stringify(proof.sessionCredentials) === JSON.stringify({ "session-a": token }),
    "Museum requests must include only the credentials for the listed proof sessions.");
  assert(runtime.withGameSessionCredentials({}).sessionCredentials === undefined,
    "Unrelated actions must not receive private credentials.");
  runtime.rememberGameSessionToken({ sessionId: "session-b", sessionToken: "malformed" });
  assert(runtime.getGameSessionToken("session-b") === null, "Invalid API credentials must not be persisted.");
  runtime = build({ getItem() { throw new Error("Storage blocked"); },
    setItem() { throw new Error("Storage full"); } });
  runtime.rememberGameSessionToken({ sessionId: "session-a", sessionToken: token });
  assert(runtime.getGameSessionToken("session-a") === token
    && runtime.getGameSessionToken("session-b") === null,
    "Credential storage failures must retain in-memory access without crashing.");
  const apiCall = source.slice(source.indexOf("async function callGameApi("),
    source.indexOf("let analyticsAccessRequest"));
  assert(apiCall.includes("withGameSessionCredentials(payload)") && apiCall.includes("rememberGameSessionToken(data)"),
    "All gameplay actions must use the shared credential transport.");
});

function buildGameApiDeadlineRuntime() {
  const source = Deno.readTextFileSync(`${javascriptRoot}/api.${javascriptExtension}`);
  const apiSource = source.slice(0, source.indexOf("let analyticsAccessRequest"));
  const deferred = () => {
    let resolve!: (value: any) => void;
    let reject!: (error: Error) => void;
    const promise = new Promise<any>((yes, no) => { resolve = yes; reject = no; });
    return { promise, resolve, reject };
  };
  const auth: ReturnType<typeof deferred>[] = [];
  const fetches: Array<ReturnType<typeof deferred> & { options: any }> = [];
  const timers = new Map<number, () => void>();
  const storage = new Map<string, string>([["visitor", "00000000-0000-4000-8000-000000000001"]]);
  const window: any = {};
  const navigator = { onLine: true };
  let nextTimer = 0;
  let now = 0;
  const runtime = new Function("env", `
    const { window, navigator, auth, fetches, timers, storage, deferred, clock, schedule } = env;
    const sb = { auth: { getSession() {
      const request = deferred(); auth.push(request); return request.promise;
    } } };
    const fetch = (url, options) => {
      const request = { ...deferred(), options }; fetches.push(request); return request.promise;
    };
    const performance = { now: clock };
    const setTimeout = schedule;
    const clearTimeout = id => timers.delete(id);
    const console = { debug() {} };
    const localStorage = { getItem: key => storage.get(key) ?? null,
      setItem: (key, value) => storage.set(key, value) };
    const PHYLOSAUR_STORAGE_KEYS = { visitorId: "visitor", sessionCredentialPrefix: "credential:" };
    const SUPABASE_ANON_KEY = "test-only-anon", GAME_API_URL = "https://example.test/game";
    const currentLocale = "en", t = key => key;
    ${apiSource}; return { callGameApi, getGameSessionToken };`)(
    { window, navigator, auth, fetches, timers, storage, deferred, clock: () => now,
      schedule(callback: () => void, delay: number) {
        assert(delay === 15000, "The game API must retain its 15-second deadline.");
        const id = ++nextTimer; timers.set(id, callback); return id;
      } },
  );
  return { ...runtime, auth, fetches, timers, storage, window, navigator, deferred,
    expire(id = [...timers.keys()][0]) { now = 15000; timers.get(id)!(); },
    authenticated: { data: { session: { access_token: "test-only-access" } } },
    response: (body: any, status = 200) => ({ status, ok: status >= 200 && status < 300,
      json: () => Promise.resolve(body) }),
  };
}

Deno.test("game API deadline covers authentication, headers and stalled JSON without accepting late results", async () => {
  const flush = async () => { for (let index = 0; index < 12; index++) await Promise.resolve(); };
  for (const stage of ["auth", "headers", "body"]) {
    for (const late of ["success", "error"]) {
      const runtime = buildGameApiDeadlineRuntime();
      const body = runtime.deferred();
      const token = `v1.${"A".repeat(43)}`;
      const data = { ok: true, sessionId: "late-session", sessionToken: token };
      const pending = runtime.callGameApi("guess", { sessionId: "original" });
      const outcome = pending.then((value: any) => ({ value, error: null }),
        (error: Error) => ({ value: null, error }));
      if (stage !== "auth") { runtime.auth[0].resolve(runtime.authenticated); await flush(); }
      if (stage === "body") {
        runtime.fetches[0].resolve({ ok: true, status: 200, json: () => body.promise }); await flush();
      }
      assert(runtime.timers.size === 1, `${stage}: the deadline was cleared before all response data arrived.`);
      runtime.navigator.onLine = false;
      runtime.expire();
      const result = await outcome;
      assert(result.error?.message === "api.timeout" && result.value === null && runtime.timers.size === 0,
        `${stage}: stalled work must release callers with a timeout, even if an adapter ignores abort.`);
      assert(!runtime.fetches.length || runtime.fetches[0].options.signal.aborted,
        "Timeout must also abort the actual network request.");
      const stale = stage === "auth" ? runtime.auth[0] : stage === "headers" ? runtime.fetches[0] : body;
      if (late === "error") stale.reject(new Error("Late transport failure"));
      else stale.resolve(stage === "auth" ? runtime.authenticated
        : stage === "headers" ? runtime.response(data) : data);
      await flush();
      const metrics = runtime.window.phylosaurPerformance;
      assert(metrics.length === 1 && !metrics[0].ok && metrics[0].durationMs === 15000
        && runtime.getGameSessionToken("late-session") === null,
        `${stage}/${late}: a late response stored credentials or reported a second result.`);
      assert(stage !== "auth" || runtime.fetches.length === 0,
        "Authentication that finishes after timeout must not send the gameplay mutation.");
    }
  }
});

Deno.test("game API retains credentials, HTTP conflict details and localized transport errors while clearing deadlines", async () => {
  const flush = async () => { for (let index = 0; index < 12; index++) await Promise.resolve(); };
  for (const scenario of ["success", "http", "invalid-json", "fetch-error", "body-abort", "offline", "auth-error"]) {
    const runtime = buildGameApiDeadlineRuntime();
    const token = `v1.${"B".repeat(43)}`;
    const data = scenario === "http" ? { ok: false, error: "Conflict", code: "SESSION_CONFLICT" }
      : { ok: true, sessionId: "current-session", sessionToken: token };
    const pending = runtime.callGameApi("state", { sessionId: "current-session" });
    const outcome = pending.then((value: any) => ({ value, error: null as any }),
      (error: any) => ({ value: null, error }));
    if (scenario === "auth-error") runtime.auth[0].reject(new Error("Auth unavailable"));
    else {
      runtime.auth[0].resolve(runtime.authenticated); await flush();
      assert(runtime.fetches[0].options.headers.Authorization === "Bearer test-only-access",
        "The deadline must not change authenticated API headers.");
      if (["fetch-error", "offline"].includes(scenario)) {
        runtime.navigator.onLine = scenario !== "offline";
        runtime.fetches[0].reject(new TypeError("Network unavailable"));
      } else if (["invalid-json", "body-abort"].includes(scenario)) {
        const error = scenario === "invalid-json" ? new SyntaxError("Invalid JSON")
          : new DOMException("Aborted response", "AbortError");
        runtime.fetches[0].resolve({ ok: true, status: 200, json: async () => { throw error; } });
      } else runtime.fetches[0].resolve(runtime.response(data, scenario === "http" ? 409 : 200));
    }
    const result = await outcome;
    assert(runtime.timers.size === 0 && runtime.window.phylosaurPerformance.length === 1,
      `${scenario}: completed requests must clear timers and record one result.`);
    if (scenario === "success") {
      assert(result.value === data && !result.error && runtime.getGameSessionToken("current-session") === token
        && runtime.window.phylosaurPerformance[0].ok, "Successful replies must still persist the matching credential.");
    } else {
      const expected = ({ http: "Conflict", "invalid-json": "api.invalidResponse",
        "fetch-error": "api.unreachable", "body-abort": "api.timeout", offline: "api.offline",
        "auth-error": "api.unreachable" } as Record<string, string>)[scenario];
      assert(result.error?.message === expected && !runtime.window.phylosaurPerformance[0].ok
        && runtime.getGameSessionToken("current-session") === null,
        `${scenario}: failure classification, metrics or credential isolation changed.`);
      if (scenario === "http") assert(result.error.status === 409 && result.error.data === data
        && result.error.serverMessage === "Conflict", "Conflict recovery needs the original structured HTTP error.");
    }
  }
});

Deno.test("overlapping game API calls use independent deadlines and do not retry a timed-out mutation", async () => {
  const runtime = buildGameApiDeadlineRuntime();
  const flush = async () => { for (let index = 0; index < 12; index++) await Promise.resolve(); };
  const first = runtime.callGameApi("guess").then(() => null, (error: Error) => error);
  const second = runtime.callGameApi("state");
  runtime.auth[0].resolve(runtime.authenticated); runtime.auth[1].resolve(runtime.authenticated); await flush();
  const ids = [...runtime.timers.keys()];
  runtime.expire(ids[0]);
  assert((await first)?.message === "api.timeout" && runtime.timers.has(ids[1])
    && runtime.fetches[0].options.signal.aborted && !runtime.fetches[1].options.signal.aborted,
    "Timing out one request must not clear or abort another request.");
  runtime.fetches[1].resolve(runtime.response({ ok: true })); await second;
  runtime.fetches[0].reject(new Error("Late aborted fetch")); await flush();
  assert(runtime.timers.size === 0 && runtime.fetches.length === 2
    && runtime.window.phylosaurPerformance.length === 2,
    "A timeout must neither replay a mutation nor leave timers or duplicate telemetry.");
});

Deno.test("session writes reject stale revisions before guesses, hints or completion can be overwritten", async () => {
  if (!edgeSourceAvailable) return;
  const source = Deno.readTextFileSync("game/index.txt");
  const helper = source.slice(source.indexOf("async function updateActiveGameSession("),
    source.indexOf("function sessionConflictResponse("))
    .replace(/: ReturnType<typeof createClient>/g, "")
    .replace(/: Record<string, unknown>/g, "");
  const update = new Function(`${helper}; return updateActiveGameSession;`)();
  const base = { id: "session", session_revision: 0, guesses: [], hint_history: [],
    hints_remaining: 2, guesses_since_hint: 2, won: false, gave_up: false,
    expires_at: "2099-01-01T00:00:00Z", updated_at: "2026-10-04T00:00:00Z" };
  const mutations = [
    { guesses: [{ nome: "Triceratops" }], guesses_since_hint: 3 },
    { hint_history: [{ cladeName: "Theropoda" }], hints_remaining: 1, guesses_since_hint: 0 },
    { gave_up: true },
    { guesses: [{ nome: "Tyrannosaurus" }], won: true },
  ];
  for (const first of mutations) {
    for (const second of mutations) {
      let record: any = structuredClone(base);
      let commits = 0;
      const admin = { from(table: string) {
        assert(table === "game_sessions", "Only the current game may be updated.");
        const filters = new Map();
        let patch: any;
        let minimumExpiry = "";
        const query: any = {
          update(value: any) { patch = value; return query; },
          eq(key: string, value: any) { filters.set(key, value); return query; },
          gt(key: string, value: string) {
            assert(key === "expires_at", "Writes must still guard session expiry.");
            minimumExpiry = value; return query;
          },
          select() { return query; },
          async maybeSingle() {
            if ([...filters].some(([key, value]) => record[key] !== value)
              || record.expires_at <= minimumExpiry) return { data: null, error: null };
            // Emulate the database trigger, including two writes in the same millisecond.
            record = { ...record, ...patch, session_revision: record.session_revision + 1,
              updated_at: base.updated_at };
            commits++;
            return { data: { id: record.id }, error: null };
          },
        };
        return query;
      } };
      const stale = structuredClone(record);
      const results = await Promise.all([update(admin, stale, first), update(admin, stale, second)]);
      assert(results.filter(Boolean).length === 1 && commits === 1 && record.session_revision === 1,
        "Two requests reading one revision must not both commit, even with identical timestamps.");
      for (const [key, value] of Object.entries(first)) {
        assert(JSON.stringify(record[key]) === JSON.stringify(value),
          `A competing request overwrote the accepted ${key}.`);
      }
      const before = JSON.stringify(record);
      assert(!await update(admin, stale, second) && JSON.stringify(record) === before,
        "Repeated stale requests must not alter guesses, hints or terminal flags.");
      if (record.won || record.gave_up) {
        assert(!await update(admin, structuredClone(record), second),
          "Even a fresh revision must not permit mutations after completion.");
      } else {
        assert(await update(admin, structuredClone(record), second),
          "An active game must remain writable after reading its newest revision.");
      }
      record = { ...base, expires_at: "2000-01-01T00:00:00Z" };
      assert(!await update(admin, structuredClone(record), first), "Expired games must not commit.");
      let rejected = false;
      try { await update(admin, { id: "session" }, first); } catch { rejected = true; }
      assert(rejected, "Missing migration must fail closed instead of bypassing revision checks.");
    }
  }
  for (const action of ["guess", "hint", "give_up"]) {
    const start = source.indexOf(`    if (action === "${action}")`);
    const end = source.indexOf("    if (action ===", start + 1);
    const handler = source.slice(start, end < 0 ? undefined : end);
    assert(handler.includes("expires_at, session_revision")
      && handler.includes("await updateActiveGameSession(admin, session,")
      && handler.includes("if (!updatedSession) return sessionConflictResponse();"),
      `${action}: the handler must use the revision it read and reject an uncommitted action.`);
    const guard = handler.indexOf("if (!updatedSession) return sessionConflictResponse();");
    assert(handler.indexOf("deferAnalyticsEvent(") > guard
      && (!handler.includes("await finalizeDailyProgress(")
        || handler.indexOf("await finalizeDailyProgress(") > guard),
      `${action}: failed writes must not report analytics or finalize Daily results.`);
    assert(handler.slice(handler.indexOf("return jsonResponse({", guard)).includes("guesses:"),
      `${action}: accepted responses must carry the authoritative guess history for stale tabs.`);
  }
  const migration = Deno.readTextFileSync("game/supabase_session_concurrency_migration.txt");
  assert(migration.includes("session_revision bigint not null default 0")
    && migration.includes("new.session_revision := old.session_revision + 1")
    && migration.includes("before update on public.game_sessions")
    && migration.includes("drop trigger if exists game_sessions_revision_guard"),
    "Revision migration must preserve saves, version all writers and support reapplication.");
});

Deno.test("playable dinosaur database preserves tree invariants", () => {
  if (!fileExists(databasePath)) {
    console.warn("phylosaur_db.json is not published in this layout; tree checks skipped.");
    return;
  }
  const database = readJson(databasePath);
  assert(Array.isArray(database) && database.length > 0, "Dinosaur database is empty.");

  const validDifficulties = new Set([
    "muito_facil",
    "facil",
    "normal",
    "dificil",
    "muito_dificil",
  ]);
  const names = new Set<string>();

  for (const dinosaur of database) {
    assert(typeof dinosaur.nome === "string" && dinosaur.nome, "A dinosaur has no name.");
    assert(!names.has(dinosaur.nome), `Duplicate playable genus: ${dinosaur.nome}`);
    names.add(dinosaur.nome);
    assert(Array.isArray(dinosaur.linhagem), `${dinosaur.nome} has no lineage.`);
    assert(dinosaur.linhagem[0] === "Dinosauria", `${dinosaur.nome} has an invalid root.`);
    assert(
      dinosaur.profundidade === dinosaur.linhagem.length,
      `${dinosaur.nome} depth does not match its lineage.`,
    );
    assert(
      new Set(dinosaur.linhagem).size === dinosaur.linhagem.length,
      `${dinosaur.nome} repeats a lineage node.`,
    );
    assert(
      validDifficulties.has(dinosaur.dificuldade),
      `${dinosaur.nome} has an invalid difficulty.`,
    );
  }

  const parents = new Map<string, Set<string>>();
  for (const dinosaur of database) {
    for (const ancestor of dinosaur.linhagem) {
      assert(!names.has(ancestor), `${ancestor} is both playable and an ancestor.`);
    }
    for (let index = 1; index < dinosaur.linhagem.length; index++) {
      const child = dinosaur.linhagem[index];
      const parent = dinosaur.linhagem[index - 1];
      if (!parents.has(child)) parents.set(child, new Set());
      parents.get(child)!.add(parent);
    }
  }

  for (const [clade, possibleParents] of parents) {
    assert(
      possibleParents.size === 1,
      `${clade} has multiple stored parents: ${[...possibleParents].join(", ")}`,
    );
  }
});

function buildWikipediaIdentityRuntime(locale = "pt-BR") {
  const source = Deno.readTextFileSync(`${javascriptRoot}/api.${javascriptExtension}`);
  const helper = source.slice(source.indexOf("async function fetchWikipediaInfo("),
    source.indexOf("async function fetchWikimediaImage("));
  const requests: Array<{ url: URL; signal: AbortSignal }> = [];
  const timers = new Map<number, () => void>();
  let nextTimer = 0;
  const state: any = { pages: { en: [], pt: [], es: [], ja: [], fr: [], de: [] }, entities: {}, handler: null };
  const fetchMock = async (rawUrl: string, options: any) => {
    const url = new URL(rawUrl);
    requests.push({ url, signal: options.signal });
    assert(url.searchParams.get("list") !== "search", "Identity must never come from full-text search ranking.");
    if (state.handler) {
      const result = state.handler(url, options);
      if (result !== undefined) return await result;
    }
    const body = url.hostname === "www.wikidata.org"
      ? { entities: state.entities }
      : { query: { pages: Object.fromEntries(state.pages[url.hostname.split(".")[0]]
        .map((page: any, index: number) => [index + 1, page])) } };
    return { ok: true, json: async () => body };
  };
  const runtime = new Function("fetch", "initialLocale", "setTimeout", "clearTimeout", "console", `
    let currentLocale = initialLocale;
    const wikipediaInfoCache = new Map();
    ${helper}; return { fetchWikipediaInfo, setLocale(value) { currentLocale = value; } };
  `)(fetchMock, locale,
    (callback: () => void, delay: number) => {
      assert(delay === 6000, "Encyclopedia requests must have a bounded deadline.");
      const id = ++nextTimer; timers.set(id, callback); return id;
    }, (id: number) => timers.delete(id), { warn() {} });
  const page = (title: string, id: string, extract = `Summary of ${title}`, extra = {}) => ({
    ns: 0, title, extract, pageprops: { wikibase_item: id }, ...extra
  });
  const entity = (name: string, rank = "normal") => ({ claims: { P225: [{
    rank, mainsnak: { snaktype: "value", datavalue: { value: name } }
  }] } });
  return { ...runtime, state, requests, timers, page, entity };
}

Deno.test("Wikipedia rejects Chupkaornis to Velociraptor and uses the confirmed English summary", async () => {
  for (const locale of ["pt-BR", "es", "ja", "fr", "de"]) {
    const runtime = buildWikipediaIdentityRuntime(locale);
    const language = ({ "pt-BR": "pt", es: "es", ja: "ja", fr: "fr", de: "de" } as Record<string, string>)[locale];
    runtime.state.pages.en = [runtime.page("Chupkaornis", "Q1", "Correct bird summary", {

  }
});

Deno.test("About loading ignores obsolete openings and preserves articles when formulas fail", async () => {
  const source = Deno.readTextFileSync(`${javascriptRoot}/screens.${javascriptExtension}`);
  const aboutSource = source.slice(source.indexOf("async function showAbout()"),
    source.indexOf("async function showHowToPlay()"));
  const deferred = () => {
    let resolve!: (value?: any) => void;
    let reject!: (error: Error) => void;
    const promise = new Promise<any>((yes, no) => { resolve = yes; reject = no; });
    return { promise, resolve, reject };
  };
  const flush = async () => { for (let index = 0; index < 8; index++) await Promise.resolve(); };
  const buildRuntime = () => {
    let route = "/";
    let html = "";
    let focused = 0;
    let rendered = 0;
    const fetches: ReturnType<typeof deferred>[] = [];
    const texts: ReturnType<typeof deferred>[] = [];
    const formulas: ReturnType<typeof deferred>[] = [];
    const content = {
      firstElementChild: null as any,
      get innerHTML() { return html; },
      set innerHTML(value: string) {
        if (this.firstElementChild) this.firstElementChild.isConnected = false;
        html = value;
        this.firstElementChild = { isConnected: true };
      },
    };
    const show = new Function("setAppRoute", "setHeaderControls", "document", "t",
      "renderAppState", "fetch", "DOMParser", "getCurrentAppRoute", "focusAppScreenHeading",
      "ensureMathRenderer", "window", "console", `${aboutSource}; return showAbout;`)(
      (value: string) => { route = value; }, () => {},
      { getElementById: () => content }, (key: string) => key, (key: string) => key,
      () => { const request = deferred(); fetches.push(request); return request.promise; },
      class {
        parseFromString(articleHtml: string) {
          return { querySelector: () => ({ innerHTML: articleHtml,
            querySelector: () => ({ remove() {} }), querySelectorAll: () => [] }) };
        }
      },
      () => route, () => { focused++; },
      () => { const request = deferred(); formulas.push(request); return request.promise; },
      { renderMathInElement: (card: any) => {
        assert(card === content.firstElementChild && card.isConnected,
          "Formula rendering targeted a detached About article.");
        rendered++;
      } }, { warn() {} },
    );
    const response = () => ({ ok: true, text: () => {
      const request = deferred(); texts.push(request); return request.promise;
    } });
    return { show, fetches, texts, formulas, response, content,
      snapshot: () => JSON.stringify({ html, focused, rendered }),
      navigate: () => { route = "/practice"; content.innerHTML = "practice"; } };
  };
  for (const stage of ["fetch", "text", "math"] as const) {
    for (const scenario of ["current", "navigate", "reopen"] as const) {
      for (const outcome of ["success", "error"] as const) {
        const runtime = buildRuntime();
        const pending = runtime.show();
        if (stage !== "fetch") { runtime.fetches[0].resolve(runtime.response()); await flush(); }
        if (stage === "math") { runtime.texts[0].resolve("article-old"); await flush(); }
        let newer: Promise<void> | undefined;
        if (scenario === "navigate") runtime.navigate();
        if (scenario === "reopen") newer = runtime.show();
        const before = runtime.snapshot();
        const request = stage === "fetch" ? runtime.fetches[0]
          : stage === "text" ? runtime.texts[0] : runtime.formulas[0];
        if (outcome === "error") request.reject(new Error("Delayed About failure"));
        else request.resolve(stage === "fetch" ? runtime.response()
          : stage === "text" ? "article-old" : undefined);
        await flush();
        if (scenario === "current" && outcome === "success") {
          if (stage === "fetch") { runtime.texts[0].resolve("article-old"); await flush(); }
          if (stage !== "math") runtime.formulas[0].resolve();
        }
        await pending;
        if (scenario !== "current" || (stage === "math" && outcome === "error")) {
          assert(runtime.snapshot() === before,
            `${stage}/${scenario}/${outcome}: a stale response or formula error replaced the screen.`);
        } else {
          assert(runtime.content.innerHTML.includes(outcome === "error" ? "about.error" : "article-old"),
            `${stage}: current About loading did not produce the expected article or error.`);
        }
        if (newer) {
          runtime.fetches[1].resolve(runtime.response()); await flush();
          runtime.texts.at(-1)!.resolve("article-new"); await flush();
          runtime.formulas.at(-1)!.resolve(); await newer;
          assert(runtime.content.innerHTML.includes("article-new"), "The newest About opening was lost.");
        }
      }
    }
  }
});

Deno.test("formula scripts share pending loads, clean up failures and retry after timeout", async () => {
  const source = Deno.readTextFileSync(`${javascriptRoot}/screens.${javascriptExtension}`);
  const loaderSource = source.slice(source.indexOf("let mathRendererPromise = null;"),
    source.indexOf("async function showAbout()"));
  const scripts: any[] = [];
  const styles: any[] = [];
  const timers = new Map<number, () => void>();
  let timerId = 0;
  const document = {
    querySelector(selector: string) {
      if (selector.startsWith("link")) return styles[0] || null;
      return scripts.find(script => selector === `script[src="${script.src}"]`) || null;
    },
    createElement(tag: string) {
      const listeners = new Map<string, Function>();
      return { tag, dataset: {}, listeners,
        addEventListener(name: string, callback: Function) { listeners.set(name, callback); },
        removeEventListener(name: string) { listeners.delete(name); },
        remove() { scripts.splice(scripts.indexOf(this), 1); },
        fire(name: string) { listeners.get(name)?.(); } };
    },
    head: { appendChild(node: any) { (node.tag === "script" ? scripts : styles).push(node); } },
  };
  const window: any = {};
  const runtime = new Function("document", "window", "setTimeout", "clearTimeout",
    `${loaderSource}; return { loadPhylosaurScript, ensureMathRenderer };`)(
    document, window,
    (callback: () => void, delay: number) => {
      assert(delay === 15000, "Formula script loading must have a bounded timeout.");
      const id = ++timerId; timers.set(id, callback); return id;
    }, (id: number) => { timers.delete(id); },
  );
  const flush = async () => { for (let index = 0; index < 8; index++) await Promise.resolve(); };
  for (const failure of ["error", "timeout"]) {
    const pending = runtime.ensureMathRenderer();
    const rejected = pending.then(() => false, () => true);
    assert(runtime.ensureMathRenderer() === pending && scripts.length === 1,
      "Concurrent formula requests must share one renderer load.");
    const failedScript = scripts[0];
    if (failure === "error") failedScript.fire("error");
    else [...timers.values()][0]();
    assert(await rejected, "Failed formula scripts must reject rather than remain pending.");
    assert(Number(scripts.length) === 0 && timers.size === 0 && failedScript.listeners.size === 0,
      "Failed scripts, event handlers and timers must be removed before a retry.");
  }
  const retry = runtime.ensureMathRenderer();
  const firstScript = scripts[0];
  assert(runtime.loadPhylosaurScript(firstScript.src) === firstScript.phylosaurLoadPromise,
    "A pending individual script must not accumulate duplicate handlers.");
  firstScript.fire("load"); await flush();
  assert(scripts.length === 2 && firstScript.dataset.loaded === "true",
    "The successful core script must be marked loaded before auto-render is requested.");
  window.renderMathInElement = () => {};
  scripts[1].fire("load"); await retry;
  await runtime.ensureMathRenderer();
  await runtime.loadPhylosaurScript(firstScript.src);
  assert(scripts.length === 2 && styles.length === 1 && timers.size === 0
    && scripts.every(script => script.listeners.size === 0),
    "Successful formula loading must reuse scripts and stylesheet without leaking timers or listeners.");
});

      remove() { if (this.parent) this.parent.children = this.parent.children.filter(node => node !== this);
        this.parent = null; }
      setAttribute(key, value) { this.attributes[key] = value; }
      getAttribute(key) { return this.attributes[key]; }
      contains(node) { return node === this || this.children.some(child => child.contains(node)); }

      addEventListener(name, callback) { this.events[name] = callback; }

            child.setAttribute(attribute[1], attribute[2]);
            if (attribute[1] === "class") child.className = attribute[2];
          }
          this.appendChild(child);
        }
      }
      get innerHTML() { return this.html || ""; }
      querySelectorAll(selector) { const all = this.children.flatMap(child => [child, ...child.querySelectorAll("*")]);
        if (selector === "*") return all;

    }

        const classes = selector.split(",").map(value => value.trim().slice(1));
        return body.children.filter(node => classes.some(name => node.className.split(" ").includes(name)));
      },
      addEventListener(name, callback) { if (name === "keydown") handlers.add(callback); },
      removeEventListener(name, callback) { if (name === "keydown") handlers.delete(callback); }
    };
    const t = (key, values) => values?.name || key;

      viewer: () => document.getElementById("image-lightbox"),
      flush() { for (const [id, callback] of [...timers]) { timers.delete(id); callback(); } },
      key(key, shiftKey = false) {
        const event = { key, shiftKey, prevented: false, stopped: false,
          preventDefault() { this.prevented = true; }, stopImmediatePropagation() { this.stopped = true; } };
        for (const callback of [...handlers]) { if (handlers.has(callback)) callback(event); if (event.stopped) break; }

function buildServiceWorkerRuntime() {
  const source = Deno.readTextFileSync(localSourceLayout ? "game/sw_service_worker.txt" : "sw.js");
  const base = "https://example.test/game/";
  const stores = new Map<string, Map<string, Response>>();
  const listeners = new Map<string, Function>();
  const failures: any = { fetchPath: "", fetchStatus: 0, putPath: "", open: false, match: false,
    optional: false, deletion: false, network: null, write: null };
  const counters = { skip: 0, claim: 0, puts: 0, fetches: [] as string[], deleted: [] as string[] };
  const key = (request: any) => new URL(typeof request === "string" ? request : request.url, base).href;
  const caches = {
    async open(name: string) {
      if (failures.open) throw new Error("Cache storage unavailable");
      if (!stores.has(name)) stores.set(name, new Map());
      return {
        async put(request: any, response: Response) {
          const path = new URL(key(request)).pathname;
          if (failures.putPath && path.endsWith(failures.putPath)) throw new Error("Quota exceeded");
          if (failures.write) await failures.write();
          counters.puts++; stores.get(name)!.set(key(request), response);
        },
        async match(request: any) {
          if (failures.match) throw new Error("Cache read refused");
          return stores.get(name)!.get(key(request))?.clone();
        },
      };
    },
    keys: async () => [...stores.keys()],
    async delete(name: string) {
      counters.deleted.push(name);
      if (failures.deletion) throw new Error("Deletion refused");
      return stores.delete(name);
    },
  };
  const self = { location: { origin: new URL(base).origin },
    addEventListener: (name: string, listener: Function) => listeners.set(name, listener),
    skipWaiting: async () => { counters.skip++; }, clients: { claim: async () => { counters.claim++; } } };
  class WorkerRequest {
    url: string; cache: string;
    constructor(path: string, options: any) { this.url = key(path); this.cache = options.cache; }
  }
  const fetch = async (request: any) => {
    const url = key(request); counters.fetches.push(url);
    if (failures.network) return await failures.network(request);
    if (failures.optional && /(?:icon|apple-touch)/.test(url)) throw new Error("Icon unavailable");
    if (failures.fetchPath && url.endsWith(failures.fetchPath)) {
      if (!failures.fetchStatus) throw new TypeError("Connection lost");
      return new Response("Missing file", { status: failures.fetchStatus });
    }
    assert(request.cache === "reload", "Installation must bypass the HTTP cache.");
    return new Response(url);
  };
  const runtime = new Function("self", "caches", "fetch", "Request", `${source};
    return { version: CACHE_VERSION, core: CORE_ASSETS, optional: OPTIONAL_ASSETS };`)(self, caches, fetch, WorkerRequest);
  const event = (name: string, request?: any) => {
    const work: Promise<any>[] = [];
    let response: Promise<Response> | undefined;
    let dispatching = true;
    listeners.get(name)!({ request,
      waitUntil(promise: Promise<any>) {
        assert(dispatching, "Background work must be registered synchronously during dispatch.");
        work.push(promise);
      },
      respondWith(promise: Promise<Response>) { response = promise; },
    });
    dispatching = false;
    return { work, response };
  };
  const seed = (name: string, path: string, text: string) => {
    if (!stores.has(name)) stores.set(name, new Map());
    stores.get(name)!.set(key(path), new Response(text));
  };
  const request = (path: string, destination = "script", mode = "same-origin", method = "GET") =>
    ({ url: key(path), destination, mode, method });
  seed("phylosaur-shell-v103", "./offline.html", "Previous offline page");
  seed("phylosaur-media", "./other.png", "Unrelated media");
  seed("another-app", "./other.js", "Unrelated app");
  return { ...runtime, stores, failures, counters, event, seed, request, key };
}

Deno.test("service worker rejects incomplete essential installs and retains the previous release", async () => {
  for (const failure of ["network", "404", "500", "write", "open"]) {
    const runtime = buildServiceWorkerRuntime();
    if (["network", "404", "500"].includes(failure)) {
      runtime.failures.fetchPath = "js/main.js?v=182";
      runtime.failures.fetchStatus = failure === "network" ? 0 : Number(failure);
    }
    if (failure === "write") runtime.failures.putPath = "style.css";
    if (failure === "open") runtime.failures.open = true;
    const install = runtime.event("install");
    const result = await Promise.allSettled(install.work);
    assert(result.some(item => item.status === "rejected") && runtime.counters.skip === 0
      && runtime.counters.claim === 0 && runtime.stores.has("phylosaur-shell-v103")
      && !runtime.stores.has(runtime.version),
      `${failure}: failed preparation must not promote a partial shell or delete the previous cache.`);
  }
  const runtime = buildServiceWorkerRuntime();
  let resolve!: (value: Response) => void;
  let settled = false;
  const delayed = new Promise<Response>(yes => { resolve = yes; });
  runtime.failures.network = async (request: any) => request.url.endsWith("js/main.js?v=182") ? delayed
    : new Response(request.url, { status: request.url.endsWith("style.css?v=182") ? 404 : 200 });
  const install = runtime.event("install");
  const work = Promise.allSettled(install.work).then(result => { settled = true; return result; });
  for (let index = 0; index < 12; index++) await Promise.resolve();
  assert(!settled && !runtime.counters.deleted.includes(runtime.version),
    "Failure cleanup must wait for late essential writers instead of deleting underneath them.");
  resolve(new Response("Last script")); await work;
  assert(!runtime.stores.has(runtime.version) && runtime.stores.has("phylosaur-shell-v103")
    && runtime.counters.skip === 0, "No late writer may leave a failed release cache behind.");
});

Deno.test("service worker delivers network responses despite cache failures and tracks background writes", async () => {
  for (const kind of ["navigation", "script", "image"]) {
    for (const failure of ["open", "read", "write", "pending-write", "http-error"]) {
      const runtime = buildServiceWorkerRuntime();
      const path = kind === "navigation" ? "./" : kind === "script" ? "./js/main.js" : "./image.png";
      const request = runtime.request(path, kind === "navigation" ? "document" : kind,
        kind === "navigation" ? "navigate" : "same-origin");
      runtime.failures.open = failure === "open";
      runtime.failures.match = failure === "read";
      runtime.failures.putPath = failure === "write" ? new URL(request.url).pathname : "";
      let resolveWrite!: () => void;
      let writeFinished = false;
      if (failure === "pending-write") {
        const pending = new Promise<void>(yes => { resolveWrite = yes; });
        runtime.failures.write = () => pending;
      }
      const response = new Response("Online response", { status: failure === "http-error" ? 500 : 200 });
      runtime.failures.network = async () => response;
      const fetched = runtime.event("fetch", request);
      assert(fetched.work.length === 1 && fetched.response, "Every intercepted network/cache task needs waitUntil.");
      const background = Promise.all(fetched.work).then(() => { writeFinished = true; });
      const delivered = await fetched.response;
      assert(delivered === response && await delivered.text() === "Online response",
        `${kind}/${failure}: cache trouble must not replace a valid online response, including HTTP error statuses.`);
      if (failure === "pending-write") {
        assert(!writeFinished, "A pending cache write must not delay responding to the page.");
        resolveWrite();
      }
      await background;
      assert(failure !== "http-error" || runtime.counters.puts === 0, "HTTP errors must never overwrite cached assets.");
    }
  }
});

Deno.test("service worker handles cache hits, background revalidation, offline misses and unrelated requests", async () => {
  const runtime = buildServiceWorkerRuntime();
  const request = runtime.request("./image.png", "image");
  runtime.seed(runtime.version, "./image.png", "Cached image");
  let resolve!: (value: Response) => void;
  let finished = false;
  runtime.failures.network = () => new Promise<Response>(yes => { resolve = yes; });
  const hit = runtime.event("fetch", request);
  const background = Promise.all(hit.work).then(() => { finished = true; });
  assert(await (await hit.response!).text() === "Cached image" && !finished,
    "Cached images must respond immediately while waitUntil retains their network refresh.");
  resolve(new Response("Updated image")); await background;
  assert(await runtime.stores.get(runtime.version)!.get(runtime.key(request))!.clone().text() === "Updated image",
    "Background refresh must finish writing its new response.");
  runtime.failures.network = async () => { throw new TypeError("Offline"); };
  runtime.seed(runtime.version, "./offline.html", "Current offline page");
  runtime.seed(runtime.version, "./js/main.js", "Cached script");
  for (const [path, destination, mode, expected, status] of [
    ["./", "document", "navigate", "Current offline page", 200],
    ["./js/main.js", "script", "same-origin", "Cached script", 200],
    ["./image.png", "image", "same-origin", "Updated image", 200],
    ["./missing.js", "script", "same-origin", "Asset unavailable offline.", 504],
  ] as const) {
    const event = runtime.event("fetch", runtime.request(path, destination, mode));
    const response = await event.response!; await Promise.all(event.work);
    assert(response.status === status && await response.text() === expected, "Offline fallback must use the current cache only.");
  }
  runtime.failures.open = true;
  for (const mode of ["navigate", "same-origin"]) {
    const event = runtime.event("fetch", runtime.request("./missing.js", "script", mode));
    assert((await event.response!).status === (mode === "navigate" ? 503 : 504),
      "Unavailable network and storage must return a controlled fallback, not reject respondWith.");
    await Promise.all(event.work);
  }
  const calls = runtime.counters.fetches.length;
  for (const request of [runtime.request("./api", "", "same-origin", "POST"),
    { ...runtime.request("./image.png", "image"), url: "https://other.test/image.png" },
    runtime.request("./unhandled", "")]) {
    const event = runtime.event("fetch", request);
    assert(!event.response && event.work.length === 0, "Non-GET, cross-origin and unrelated requests must pass through.");
  }
