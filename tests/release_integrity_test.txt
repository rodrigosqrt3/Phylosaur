function fileExists(path: string) {
  try {
    return Deno.statSync(path).isFile;
  } catch {
    return false;
  }
}

const localSourceLayout = fileExists("game/js/config.txt");
const gameRoot = localSourceLayout ? "game" : ".";
const javascriptRoot = localSourceLayout ? "game/js" : "js";
const javascriptExtension = localSourceLayout ? "txt" : "js";
const javascriptNames = [
  "config", "state", "api", "autocomplete", "db", "auth",

function assert(condition: unknown, message: string): asserts condition {
  if (!condition) throw new Error(message);
}

function readJson(path: string) {
  return JSON.parse(Deno.readTextFileSync(path));
}

Deno.test("browser JavaScript parses individually and in production order", () => {
  let combined = "";
  for (const path of javascriptSources) {
    const source = Deno.readTextFileSync(path);
    try {
      new Function(source);
    } catch (error) {
      throw new Error(`${path} does not parse: ${error}`);
    }
    combined += `\n${source}`;
  }
  new Function(combined);
});

Deno.test("playable dinosaur database preserves tree invariants", () => {
  if (!fileExists(databasePath)) {
    console.warn("phylosaur_db.json is not published in this layout; tree checks skipped.");
    return;
  }
  const database = readJson(databasePath);
  assert(Array.isArray(database) && database.length > 0, "Dinosaur database is empty.");

  const validDifficulties = new Set([
    "muito_facil",
    "facil",
    "normal",
    "dificil",
    "muito_dificil",
  ]);
  const names = new Set<string>();

  for (const dinosaur of database) {
    assert(typeof dinosaur.nome === "string" && dinosaur.nome, "A dinosaur has no name.");
    assert(!names.has(dinosaur.nome), `Duplicate playable genus: ${dinosaur.nome}`);
    names.add(dinosaur.nome);
    assert(Array.isArray(dinosaur.linhagem), `${dinosaur.nome} has no lineage.`);
    assert(dinosaur.linhagem[0] === "Dinosauria", `${dinosaur.nome} has an invalid root.`);
    assert(
      dinosaur.profundidade === dinosaur.linhagem.length,
      `${dinosaur.nome} depth does not match its lineage.`,
    );
    assert(
      new Set(dinosaur.linhagem).size === dinosaur.linhagem.length,
      `${dinosaur.nome} repeats a lineage node.`,
    );
    assert(
      validDifficulties.has(dinosaur.dificuldade),
      `${dinosaur.nome} has an invalid difficulty.`,
    );
  }

  const parents = new Map<string, Set<string>>();
  for (const dinosaur of database) {
    for (const ancestor of dinosaur.linhagem) {
      assert(!names.has(ancestor), `${ancestor} is both playable and an ancestor.`);
    }
    for (let index = 1; index < dinosaur.linhagem.length; index++) {
      const child = dinosaur.linhagem[index];
      const parent = dinosaur.linhagem[index - 1];
      if (!parents.has(child)) parents.set(child, new Set());
      parents.get(child)!.add(parent);
    }
  }

  for (const [clade, possibleParents] of parents) {
    assert(
      possibleParents.size === 1,
      `${clade} has multiple stored parents: ${[...possibleParents].join(", ")}`,
    );
  }
});

Deno.test("published paleodata is source-reviewed and belongs to playable genera", () => {
  const playableNames = fileExists(databasePath)
    ? new Set(readJson(databasePath).map((dinosaur: { nome: string }) => dinosaur.nome))
    : null;
  const paleodata = readJson(assetPath("phylosaur_paleodata.json"));
  const taxa = paleodata.taxa ?? {};

  for (const [name, record] of Object.entries<any>(taxa)) {
    if (playableNames) {
      assert(playableNames.has(name), `Paleodata contains unknown genus: ${name}`);
    }
    assert(record.verification?.status === "source_verified", `${name} is not source-verified.`);
    assert(record.period && record.age_text, `${name} has incomplete age information.`);
    assert(Array.isArray(record.countries) && record.countries.length, `${name} has no country.`);
    assert(Array.isArray(record.continents) && record.continents.length, `${name} has no continent.`);
    assert(Array.isArray(record.formations) && record.formations.length, `${name} has no formation.`);
    assert(
      Array.isArray(record.verification?.sources) && record.verification.sources.length,
      `${name} has no reviewed source.`,
    );
    for (const source of record.verification.sources) {
      assert(source.citation, `${name} has a source without citation.`);
      assert(/^https?:\/\//.test(source.url), `${name} has an invalid source URL.`);
    }
  }
});

Deno.test("autocomplete ranks exact names, prefixes and interior matches before limiting", () => {
  const source = Deno.readTextFileSync(`${javascriptRoot}/autocomplete.${javascriptExtension}`);
  const rank = new Function(`${source}; return getRankedDinosaurSuggestions;`)();
  const candidates = ["Yutyrannus", "Styracosaurus", "Tyrannotitan", "Tyrannosaurus"]
    .map(nome => ({ nome }));
  const original = JSON.stringify(candidates);
  const names = (matches: { nome: string }[]) => matches.map(match => match.nome).join(",");
  assert(
    names(rank(candidates, "  TYR  ")) === "Tyrannosaurus,Tyrannotitan,Styracosaurus,Yutyrannus",
    "Prefixes must precede interior matches, with alphabetical ties and case-insensitive input.",
  );
  assert(
    names(rank([...candidates].reverse(), "tyr")) === names(rank(candidates, "tyr")),
    "Suggestion order must not depend on catalog order.",
  );
  assert(JSON.stringify(candidates) === original, "Ranking must not reorder the game database.");
  assert(
    names(rank(candidates, "tyr", new Set(["tyrannosaurus"]))) === "Tyrannotitan,Styracosaurus,Yutyrannus",
    "Already-guessed names must stay excluded.",
  );
  const fixtures = [
    ...Array.from({ length: 12 }, (_, index) => ({ nome: `Other${index}Match` })),
    { nome: "MatchPrefix" }, { nome: "Match" },
  ];
  const limited = rank(fixtures, "match");
  assert(limited.length === 8, "Suggestions must remain limited to eight.");
  assert(limited[0].nome === "Match" && limited[1].nome === "MatchPrefix",
    "Exact and prefix matches must not be discarded before ranking.");
  for (const query of ["", " ", "t", "zzzzzz"]) {
    assert(rank(candidates, query).length === 0, `Unexpected suggestions for '${query}'.`);
  }
});

Deno.test("guess input displays ranked suggestions and submits the typed name", () => {
  const source = Deno.readTextFileSync(`${javascriptRoot}/autocomplete.${javascriptExtension}`);
  const handlers: Record<string, Function> = {};
  let guesses = 0;
  const status = { textContent: "" };
  const input = {
    value: "tyr",
    setAttribute() {}, removeAttribute() {},
    addEventListener: (event: string, handler: Function) => { handlers[event] = handler; },
  };
  const suggestions = {
    innerHTML: "", style: { display: "none" },
    parentElement: { querySelector: () => status },
    setAttribute() {}, querySelector: () => null, querySelectorAll: () => [],
  };
  const initialize = new Function(
    "document", "window", "database", "guessedNames", "makeGuess",
    `${source}; return initializeAutocomplete;`,
  )(
    {
      getElementById: (id: string) => id === "dino-input" ? input : suggestions,
      addEventListener() {}, querySelector: () => null,
    },
    {},
    ["Yutyrannus", "Tyrannotitan", "Tyrannosaurus"].map(nome => ({ nome })),
    new Set(["tyrannotitan"]),
    () => { guesses++; },
  );
  initialize();
  handlers.input.call(input);
  assert(suggestions.innerHTML.includes("Tyrannosaurus") && !suggestions.innerHTML.includes("Tyrannotitan"),
    "Rendered suggestions must exclude previous guesses.");
  assert(suggestions.innerHTML.indexOf("Tyrannosaurus") < suggestions.innerHTML.indexOf("Yutyrannus"),
    "The input must render prefix matches first.");
  assert(status.textContent === "2 dinosaur suggestions available.", "Suggestion count is incorrect.");
  input.value = "Yutyrannus";
  handlers.keydown({ key: "Enter", preventDefault() {} });
  assert(guesses === 1 && input.value === "Yutyrannus", "Ranking must not replace a typed guess on Enter.");
  handlers.keydown({ key: "Escape", preventDefault() {} });
  assert(suggestions.style.display === "none", "Escape must still dismiss suggestions.");
});

Deno.test("Museum dismissal animates once and restores scroll and card focus", async () => {
  const screensSource = Deno.readTextFileSync(`${javascriptRoot}/screens.${javascriptExtension}`);
  const closeSource = screensSource.slice(
    screensSource.indexOf("async function closeMuseumEntry("),
    screensSource.indexOf("function openMuseumImageViewer("),
  );
  for (const reducedMotion of [false, true]) {
    for (const depth of [0, 1]) {
      let backCalls = 0;
      let replacedRoute = "";
      let scroll: any;
      let focus: any;
      let duration = -1;
      let animated = false;
      let currentOverlay: any;
      const body = { style: { overflow: "hidden" } };
      const overlay = {
        museumReturnState: {
          scrollX: 0, scrollY: 1460, overflow: "auto",
          card: { isConnected: true, focus: (options: any) => { focus = options; } },
        },
        classList: { add: (name: string) => { animated = name === "is-closing"; } },
        remove: () => { currentOverlay = null; },
      };
      currentOverlay = overlay;
      const dismiss = new Function(
        "document", "window", "getCurrentAppRoute", "setAppRoute", "setTimeout",
        `let activeMuseumEntryMedia = null; let museumEntryEscapeHandler = null;
         ${closeSource}; return dismissMuseumEntry;`,
      )(
        { body, getElementById: (id: string) => id === "museum-entry-overlay" ? currentOverlay : null },
        {
          matchMedia: () => ({ matches: reducedMotion }),
          scrollTo: (options: any) => { scroll = options; },
          history: { state: { phylosaurDepth: depth }, back: () => { backCalls++; } },
        },
        () => "/museum/Triceratops",
        (route: string) => { replacedRoute = route; },
        (resolve: () => void, delay: number) => { duration = delay; queueMicrotask(resolve); },
      );
      await Promise.all([dismiss(), dismiss()]);
      assert(animated && duration === (reducedMotion ? 0 : 180), "Motion preference was ignored.");
      assert(currentOverlay === null && body.style.overflow === "auto", "Scroll lock was not released.");
      assert(scroll?.top === 1460 && scroll?.behavior === "instant", "Gallery position was lost.");
      assert(focus?.preventScroll === true, "Focus must return without scrolling the card into view.");
      assert(backCalls === depth, "Repeated close events must not navigate back twice.");
      assert(replacedRoute === (depth ? "" : "/museum"), "Direct links must close to the Museum.");
    }
  }
});
